cmake_minimum_required(VERSION 3.24)
project(event-log-triage VERSION 1.0.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Windows and MSVC are required")
endif()
option(TRIAGE_ANALYZE "Enable MSVC analysis" OFF)
option(TRIAGE_ASAN "Enable address sanitizer" OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(triage src/cli.cpp src/query.cpp src/grouping.cpp src/correlation.cpp
  src/json_formatter.cpp src/human_formatter.cpp src/help.cpp src/acquisition.cpp
  src/xml_parser.cpp src/message_formatter.cpp src/unicode_time.cpp)
target_include_directories(triage PUBLIC src)
target_compile_definitions(triage PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_link_libraries(triage PUBLIC wevtapi xmllite shlwapi ole32)
add_executable(event-log-triage src/main.cpp)
target_link_libraries(event-log-triage PRIVATE triage)
add_executable(triage-tests tests/tests.cpp)
target_link_libraries(triage-tests PRIVATE triage advapi32)
add_executable(triage-nonadmin tests/nonadmin.cpp)
target_compile_definitions(triage-nonadmin PRIVATE UNICODE _UNICODE)
target_link_libraries(triage-nonadmin PRIVATE advapi32)
foreach(t triage event-log-triage triage-tests triage-nonadmin)
  target_compile_options(${t} PRIVATE /W4 /WX /utf-8 /permissive- /EHsc)
  if(TRIAGE_ANALYZE)
    target_compile_options(${t} PRIVATE /analyze)
  endif()
  if(TRIAGE_ASAN)
    target_compile_options(${t} PRIVATE /fsanitize=address /Zi)
    target_link_options(${t} PRIVATE /INCREMENTAL:NO)
  endif()
endforeach()
enable_testing()
add_test(NAME synthetic COMMAND triage-tests)
add_test(NAME system COMMAND event-log-triage --log System --since 2h --limit 2 --json)
add_test(NAME application COMMAND event-log-triage --log Application --since 2h --limit 2 --json)
add_test(NAME unavailable COMMAND triage-tests --unavailable)
add_test(NAME nonadmin COMMAND triage-nonadmin $<TARGET_FILE:triage-tests>)
add_test(NAME bounds COMMAND triage-tests --bounds)
set_tests_properties(synthetic system application unavailable nonadmin bounds PROPERTIES TIMEOUT 60)
install(TARGETS event-log-triage RUNTIME DESTINATION bin)
install(FILES README.md LICENSE DESTINATION share/event-log-triage)
install(FILES docs/schema-1.0.json DESTINATION share/event-log-triage)
